{{/* 页内图：资源目录图出响应式 WebP；外链原样。 */}} {{- $destination := .Destination -}} {{- $path := strings.TrimPrefix "./" $destination -}} {{- with .Page.Resources.GetMatch $path -}} {{- $image := . -}} {{- if in (slice "jpeg" "png" "tiff" "webp") $image.MediaType.SubType -}} {{- $targetWidth := $image.Width -}} {{- if gt $targetWidth 1600 -}} {{- $targetWidth = 1600 -}} {{- end -}} {{- $fallback := $image.Resize (printf "%dx q82" $targetWidth) -}} {{- $srcset := slice -}} {{- range slice 640 960 1440 -}} {{- if lt . $targetWidth -}} {{- $variant := $image.Resize (printf "%dx webp q75" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $variant.RelPermalink $variant.Width) -}} {{- end -}} {{- end -}} {{- $webp := $image.Resize (printf "%dx webp q75" $targetWidth) -}} {{- $srcset = $srcset | append (printf "%s %dw" $webp.RelPermalink $webp.Width) -}} {{- else -}} {{- end -}} {{- else -}} {{- end -}}

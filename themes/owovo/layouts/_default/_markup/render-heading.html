"class" "anchor-icon") -}} {{- $anchorContent = . -}} {{- end -}} {{- end -}} {{- if $page.Site.Params.enableAnchorLink -}} {{ $anchorContent }} {{- else -}} {{ $anchorContent }} {{- end -}} {{- end -}} {{- if $linkToTOC -}} {{ .Text | safeHTML }} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
